int main() {

	printf("pscore의 값: %p\n", pscore);
	printf("=====================================\n");
	printf("pscore의 주소: %p\n", &pscore);
	printf("ppscore의 값: %p\n", ppscore);	
	printf("=====================================\n");
	printf("ppscore의 주소: %p\n", &ppscore);
	printf("pppscore의 값: %p\n", pppscore);

	return 0;
}
