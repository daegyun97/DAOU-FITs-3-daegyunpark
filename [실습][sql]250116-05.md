```
-- 실습과제 7
-- 1번
SELECT B.NAME, C.NAME, SUM(L.AMOUNT) AS "대출합계"
FROM BRANCHES B
LEFT JOIN LOANS L ON B.BRANCH_ID = L.BRANCH_ID 
LEFT JOIN  CUSTOMERS C ON C.CUSTOMER_ID = L.CUSTOMER_ID
GROUP BY B.NAME, C.NAME;

-- 2번
SELECT C.NAME, C.EMAIL FROM CUSTOMERS C
INNER JOIN 
(SELECT DISTINCT CUSTOMER_ID FROM ACCOUNTS WHERE BALANCE >50000) A
ON C.CUSTOMER_ID = A.CUSTOMER_ID;

-- 3번
SELECT L.AMOUNT, C.NAME FROM LOANS L
LEFT JOIN CUSTOMERS C ON L.CUSTOMER_ID = C.CUSTOMER_ID
WHERE L.STATUS = 'PENDING';
```
