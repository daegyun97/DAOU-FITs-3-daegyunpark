package booksearch.controller;

import booksearch.service.BookService;
import booksearch.service.BookServiceImpl;
import booksearch.VO.BookVO;
import java.net.URL;
import java.util.ResourceBundle;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.Button;
import javafx.scene.control.ContextMenu;
import javafx.scene.control.MenuItem;
import javafx.scene.control.TableColumn;
import javafx.scene.control.TableView;
import javafx.scene.control.TextField;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.control.cell.TextFieldTableCell;

public class BookController implements Initializable {
  @FXML
  private TableView<BookVO> tableView;
  @FXML
  private TableColumn<BookVO, String> isbn;
  @FXML
  private TableColumn<BookVO, String> title;
  @FXML
  private TableColumn<BookVO, Integer> price;
  @FXML
  private TableColumn<BookVO, String> author;
  @FXML
  private Button searchBtn;
  @FXML
  private Button deleteBtn;
  @FXML
  private TextField input;

  BookService bookService = new BookServiceImpl();

  public BookController() throws Exception {
  }

  @Override
  public void initialize(URL url, ResourceBundle resourceBundle) {
    isbn.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
    title.setCellValueFactory(new PropertyValueFactory<>("btitle"));
    price.setCellValueFactory(new PropertyValueFactory<>("bprice"));
    author.setCellValueFactory(new PropertyValueFactory<>("bauthor"));

    //수정
    tableView.setOnMouseClicked(event -> {
      tableView.setEditable(true);
    });
    author.setCellFactory(TextFieldTableCell.forTableColumn());

    author.setOnEditCommit(event -> { // 수정된 값 저장
      BookVO bookVO = event.getRowValue();
      bookVO.setBauthor(event.getNewValue());
      try {
        bookService.updateBook(event.getRowValue());
      } catch (Exception e) {
        throw new RuntimeException(e);
      }
    });


    // 검색
    searchBtn.setOnAction(event->{
      String keyWord = input.getText();
      tableView.setItems(bookService.getBooks(keyWord));
    });

    // 삭제
    ContextMenu contextMenu = new ContextMenu();
    MenuItem deleteItem = new MenuItem("삭제");
    contextMenu.getItems().add(deleteItem);
    tableView.setContextMenu(contextMenu);

    deleteItem.setOnAction(e -> {
      BookVO selectedBookVO = tableView.getSelectionModel().getSelectedItem();
      if (selectedBookVO != null) {
        try {
          bookService.deleteBook(selectedBookVO.getBisbn());
        } catch (Exception ex) {
          ex.printStackTrace();
        }
        tableView.getItems().remove(selectedBookVO);
      }
    });



  }
}

package booksearch.service;

import booksearch.VO.BookVO;
import booksearch.dao.BookDAO;
import booksearch.dao.BookDAOImpl;
import java.sql.Connection;
import javafx.collections.ObservableList;

public class BookServiceImpl implements BookService {
  ConnectionMaker cm = new OracleConnector();
  BookDAO bookDAO = new BookDAOImpl(cm);

  public BookServiceImpl() {}

  @Override
  public ObservableList<BookVO> getBooks(String keyWord)  {
    ObservableList<BookVO> books = null;
    try{
      Connection con = cm.getConnection(); // Connection 획득
      con.setAutoCommit(false);
      books = bookDAO.findAll("%"+keyWord+"%");
      con.commit();
    } catch (Exception e) {
      e.printStackTrace();
    }
    return books;
  }

  @Override
  public void updateBook(BookVO bookVO)  {
    try{
      bookDAO.updateById(bookVO);
    }catch (Exception e) {
      e.printStackTrace();
    }
  }

  @Override
  public void deleteBook(String bisbn)  {
    try{
      bookDAO.deleteByKey(bisbn);
    }catch (Exception e) {
      e.printStackTrace();
    }
  };

}

package booksearch.dao;

import booksearch.VO.BookVO;
import booksearch.service.ConnectionMaker;
import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;


public class BookDAOImpl implements BookDAO{

  private ConnectionMaker connectionMaker;

  public BookDAOImpl(ConnectionMaker connectMaker) {
    this.connectionMaker = connectMaker;
  }

  @Override
  public ObservableList<BookVO> findAll(String keyWord) throws Exception {
    ObservableList<BookVO> bookVOS = FXCollections.observableArrayList();
    String sql = "select bisbn, btitle, bauthor, bprice from book"
        + " where btitle like ?";

    Connection con =  connectionMaker.getConnection();
    PreparedStatement ps = con.prepareStatement(sql);
    ps = con.prepareStatement(sql);
    ps.setString(1, keyWord);
    ResultSet rs = ps.executeQuery();
    while (rs.next()) {
      bookVOS.add(new BookVO(rs.getString("bisbn"),rs.getString("btitle"),
          rs.getInt("bprice"),rs.getString("bauthor")));
    }
    rs.close();
    ps.close();
    con.close();
    return bookVOS;
  }

  @Override
  public void updateById(BookVO bookVO) throws Exception {
    String sql = "UPDATE BOOK "
        + "SET bauthor = ?, "
        + "btitle = ?, "
        + "bprice = ? "
        + "WHERE bisbn = ?";
    Connection con =  connectionMaker.getConnection();
    PreparedStatement ps = con.prepareStatement(sql);
    ps = con.prepareStatement(sql);
    ps.setString(1, bookVO.getBauthor());
    ps.setString(2, bookVO.getBtitle());
    ps.setInt(3, bookVO.getBprice());
    ps.setString(4, bookVO.getBisbn());
    ps.executeUpdate();
    ps.close();
    con.close();
  }

  @Override
  public void deleteByKey(String bisbn) throws Exception {
    String sql = "delete from book where bisbn=?";
    Connection con =  connectionMaker.getConnection();
    PreparedStatement ps = con.prepareStatement(sql);
    ps = con.prepareStatement(sql);
    ps.setString(1,bisbn);
    ps.executeUpdate();
    ps.close();
    con.close();
  }

}
