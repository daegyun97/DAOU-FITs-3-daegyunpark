package first;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.util.Scanner;

public class SecondConnect {

  static final String DRIVER = "oracle.jdbc.driver.OracleDriver";
  static final String IP = "127.0.0.1";
  static final String PORT = "1521";
  static final String SID = "xe";
  static final String USER = "C##TEST1";
  static final String PW = "0000";

  public static void main(String[] args) throws Exception {
    //1. JDBC driver loading
    Class.forName(DRIVER);

    //2.connect to database
    String URL = "jdbc:oracle:thin:@" + IP + ":" + PORT + "/" + SID;
    Connection con = DriverManager.getConnection(URL, USER, PW);

    //3.create sql statement
    // 3-1. 일반 statement 잘 사용하지 않음
    // 3-2. Callable statement => stored procedure 호출
    Scanner sc = new Scanner(System.in);
    System.out.println("검색어를 입력하시오");
    String keyWord = sc.nextLine();
    String sql = "select bisbn, btitle, bauthor, bprice from book "
        + "where btitle like ?"; //인파라미터 사용
    PreparedStatement ps = con.prepareStatement(sql);//3-3 주로 사용
    ps.setString(1, "%" + keyWord + "%");

    //4.execute
    // executeUpdate() - select 제외한 구문 return 값 변경된 row 수
    // executeQuery() - select 문
    ResultSet rs = ps.executeQuery();

    //5.result
    while (rs.next()) {
      System.out.print(rs.getString("bisbn") + " ");
      System.out.print(rs.getString("btitle") + " ");
      System.out.print(rs.getString("bauthor") + " ");
      System.out.println(rs.getString("bprice"));
    }

    //6.disconnect database
    rs.close();
    ps.close();
    con.close();
  }
}
