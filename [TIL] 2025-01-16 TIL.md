### 학습하는데 어떤 문제에 부딪혔는지
  ```
  SELECT * FROM(
      SELECT C.NAME, SUM(L.AMOUNT) AS TOTAL FROM LOANS L
      INNER JOIN CUSTOMERS C ON C.CUSTOMER_ID = L.CUSTOMER_ID
      GROUP BY C.NAME
      ORDER BY TOTAL
  )
  WHERE ROWNUM<6;
  ```

    GROUP BY DEPARTMENT_ID
    ORDER BY CNT)

SELECT JOB_ID, AVG(SALARY), 
    CASE
        WHEN AVG(SALARY) >= 8000 THEN 'HIGH'
        WHEN AVG(SALARY) >= 6000 THEN 'MEDIUM'
        ELSE 'LOW'
        END AS GRADE
FROM EMPLOYEES
GROUP BY JOB_ID
HAVING AVG(SALARY)>5000;

-- 18
SELECT C.NAME, SUM(T.AMOUNT) FROM CUSTOMERS C
INNER JOIN ACCOUNTS A ON A.CUSTOMER_ID = C.CUSTOMER_ID
INNER JOIN TRANSACTIONS T ON T.ACCOUNT_ID = A.ACCOUNT_ID
GROUP BY C.NAME
ORDER BY TO_NUMBER(REGEXP_REPLACE(C.NAME, '[^0-9]'));


-- 19

WHERE DEPARTMENT_ID IS NOT NULL

SELECT * FROM(

    INNER JOIN CUSTOMERS C ON C.CUSTOMER_ID = L.CUSTOMER_ID

    ORDER BY TOTAL
