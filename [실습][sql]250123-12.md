-- Q1
SELECT B.NAME, SUM(L.AMOUNT) AS BRANCH, ROUND((SELECT AVG(AMOUNT) FROM LOANS WHERE STATUS = 'APPROVED'),2) AS TOTAL 
FROM LOANS L
LEFT JOIN BRANCHES B ON B.BRANCH_ID = L.BRANCH_ID
WHERE L.STATUS = 'APPROVED'
GROUP BY B.NAME
HAVING SUM(L.AMOUNT)<(SELECT AVG(AMOUNT) FROM LOANS WHERE STATUS = 'APPROVED');

SELECT * FROM
(
    SELECT DISTINCT B.NAME, 
        SUM(L.AMOUNT) OVER(PARTITION BY B.NAME) AS BR, 
        ROUND(AVG(L.AMOUNT) OVER (),2) AS TT
    FROM LOANS L
    LEFT JOIN BRANCHES B ON B.BRANCH_ID = L.BRANCH_ID
    WHERE L.STATUS = 'APPROVED'
)
WHERE BR<TT;


-- Q2
SELECT E1.EMPLOYEE_ID, E1.NAME, NVL(E2.NAME, 'NO MANAGER'), NVL(D.DEPARTMENT_NAME, 'NO DEPARMENT')
FROM EMPLOYEES E1
LEFT JOIN EMPLOYEES E2 ON E1.MANAGER_ID = E2.EMPLOYEE_ID
LEFT JOIN DEPARTMENTS D ON E2.DEPARTMENT_ID = D.DEPARTMENT_ID;

-- Q3
SELECT EMPLOYEE_ID, NAME, SALARY, 
    ROUND(SALARY/SUM(SALARY) OVER()*100,2) AS PER
FROM EMPLOYEES
WHERE SALARY IS NOT NULL
ORDER BY PER DESC;

-- Q4
SELECT * FROM
(
    SELECT A.*,
        DENSE_RANK() OVER(ORDER BY M DESC) AS RANKING
    FROM
    (
        SELECT DISTINCT ACCOUNT_ID,
            MAX(AMOUNT) OVER(PARTITION BY ACCOUNT_ID) AS M
        FROM TRANSACTIONS
        ORDER BY M DESC
    ) A
)
WHERE RANKING<=5;

SELECT ACCOUNT_ID, MAX(AMOUNT),
    DENSE_RANK() OVER (ORDER BY MAX(AMOUNT) DESC)
FROM TRANSACTIONS
GROUP BY ACCOUNT_ID;

-- Q5
SELECT DEPARTMENT_ID, MAX(SALARY) - MIN(SALARY) AS M
FROM EMPLOYEES E
WHERE DEPARTMENT_ID IS NOT NULL
GROUP BY DEPARTMENT_ID
ORDER BY DEPARTMENT_ID;

-- Q6
SELECT EMPLOYEE_ID, NAME, SALARY, 
    ROUND(AVG(SALARY) OVER(PARTITION BY DEPARTMENT_ID)) AS TOTAL, 
    ROUND(SALARY - AVG(SALARY) OVER(PARTITION BY DEPARTMENT_ID)) AS DIF
FROM EMPLOYEES
WHERE SALARY IS NOT NULL;

-- Q7
SELECT * FROM
(
    SELECT LTRIM(SYS_CONNECT_BY_PATH(NAME, '>'), '>') AS PATH
    FROM EMPLOYEES
    START WITH NAME = 'Employee 241'
    CONNECT BY EMPLOYEE_ID = PRIOR MANAGER_ID
    ORDER BY PATH DESC
)
WHERE ROWNUM<2;

-- Q8
SELECT * FROM
(
    SELECT NAME, SALARY
    FROM EMPLOYEES
    WHERE SALARY IS NOT NULL
    ORDER BY SALARY DESC, NAME
)
WHERE ROWNUM<=10;

-- Q9
SELECT NVL(TO_CHAR(EXTRACT(YEAR FROM HIRE_DATE)),'ALL YEARS'), SUM(SALARY)
FROM EMPLOYEES
GROUP BY ROLLUP(EXTRACT(YEAR FROM HIRE_DATE)); 

-- Q10
SELECT D.DEPARTMENT_NAME, E.JOB_ID, SUM(E.SALARY)
FROM EMPLOYEES E
LEFT JOIN DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
WHERE E.DEPARTMENT_ID IS NOT NULL
GROUP BY GROUPING SETS(D.DEPARTMENT_NAME, E.JOB_ID,())
ORDER BY D.DEPARTMENT_NAME, E.JOB_ID;

-- Q11
SELECT DEPARTMENT_ID, NAME, SALARY,
    SUM(SALARY) OVER(PARTITION BY DEPARTMENT_ID ORDER BY SALARY RANGE BETWEEN 100 PRECEDING AND 100 FOLLOWING) AS RESULT
FROM EMPLOYEES;


-- Q12
SELECT NAME, SALARY,
    CASE WHEN AVG(SALARY) OVER(PARTITION BY JOB_ID)>SALARY THEN '직무평균이하'
    ELSE '직무평균이상'
    END AS JOB_AVG,
    CASE WHEN AVG(SALARY) OVER() >SALARY THEN '전체평균이하'
    ELSE '전체평균이상'
    END AS TOTAL_AVG
FROM EMPLOYEES
WHERE SALARY IS NOT NULL;
