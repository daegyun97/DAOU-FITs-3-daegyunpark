### 오늘 내가 새롭게 배운 것은 무엇인지?
 - 그룹 함수
   ```
    SELECT JOB_ID, COUNT(*) AS CNT
    FROM EMPLOYEES
    GROUP BY JOB_ID
    ORDER BY JOB_ID;
    
    SELECT JOB_ID, COUNT(*)AS CNT
    FROM EMPLOYEES
    GROUP BY ROLLUP(JOB_ID)
    ORDER BY JOB_ID;
    
    SELECT JOB_ID, DEPARTMENT_ID, COUNT(*)
    FROM EMPLOYEES
    WHERE DEPARTMENT_ID IS NOT NULL
    GROUP BY JOB_ID, DEPARTMENT_ID;
    
    SELECT JOB_ID, DEPARTMENT_ID, COUNT(*)
    FROM EMPLOYEES
    WHERE DEPARTMENT_ID IS NOT NULL
    GROUP BY ROLLUP (JOB_ID, DEPARTMENT_ID)
    ORDER BY JOB_ID, DEPARTMENT_ID;
    
    SELECT JOB_ID, COUNT(*) AS CNT
    FROM EMPLOYEES
    GROUP BY CUBE(JOB_ID)
    ORDER BY JOB_ID;
    
    SELECT JOB_ID, DEPARTMENT_ID, COUNT(*) AS CNT
    FROM EMPLOYEES
    WHERE DEPARTMENT_ID IS NOT NULL
    GROUP BY CUBE(JOB_ID, DEPARTMENT_ID)
    ORDER BY JOB_ID, DEPARTMENT_ID;
    
    SELECT JOB_ID, COUNT(*) AS CNT 
    FROM EMPLOYEES
    GROUP BY GROUPING SETS(JOB_ID)
    ORDER BY JOB_ID;
    
    SELECT JOB_ID, DEPARTMENT_ID, COUNT(*)AS CNT
    FROM EMPLOYEES
    WHERE DEPARTMENT_ID IS NOT NULL
    GROUP BY GROUPING SETS(JOB_ID, DEPARTMENT_ID,())
    ORDER BY JOB_ID, DEPARTMENT_ID;
    
    SELECT JOB_ID,NULL AS DEPARMENT_ID, COUNT(*) AS CNT
    FROM EMPLOYEES
    GROUP BY JOB_ID
    UNION ALL
    SELECT NULL,TO_CHAR(DEPARTMENT_ID), COUNT(*) AS CNT
    FROM EMPLOYEES
    GROUP BY DEPARTMENT_ID;
    
    SELECT JOB_ID, COUNT(*) AS CNT, GROUPING(JOB_ID)
    FROM EMPLOYEES
    GROUP BY ROLLUP(JOB_ID)
    ORDER BY JOB_ID;
    
    SELECT JOB_ID, DEPARTMENT_ID, COUNT(*) AS CNT, GROUPING(JOB_ID), GROUPING(DEPARTMENT_ID)
    FROM EMPLOYEES
    WHERE DEPARTMENT_ID IS NOT NULL
    GROUP BY ROLLUP(JOB_ID, DEPARTMENT_ID)
    ORDER BY JOB_ID, DEPARTMENT_ID;
    
    SELECT CASE WHEN GROUPING(JOB_ID) = 0 THEN JOB_ID
                WHEN GROUPING(JOB_ID) = 1 THEN '총합'
            END JOB_ID,
            COUNT(*) AS CNT, GROUPING(JOB_ID)
    FROM EMPLOYEES
    GROUP BY ROLLUP (JOB_ID)
    ORDER BY JOB_ID;
   ```
 - 뷰
   ```
    CREATE OR REPLACE VIEW EMPLOYEES_VIEW AS
        SELECT EMPLOYEE_ID, NAME, JOB_ID, DEPARTMENT_ID, SALARY
        FROM EMPLOYEES
        WHERE MOD(DEPARTMENT_ID,2) = 0;
    
    SELECT * FROM EMPLOYEES_VIEW WHERE EMPLOYEE_ID = 101;
    
    SELECT SALARY FROM EMPLOYEES WHERE EMPLOYEE_ID = 101;
    UPDATE EMPLOYEES_VIEW SET SALARY = 6000
    WHERE EMPLOYEE_ID = 101;
   ```
