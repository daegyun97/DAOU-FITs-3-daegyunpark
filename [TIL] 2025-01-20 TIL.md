 - 집합 연산자(UNION, MINUS, INTERSECT)
   ```
    -- 관련 실습
    -- Q1 
    SELECT D.DEPARTMENT_NAME, E.JOB_ID, ROUND(AVG(E.SALARY),2) AS AVG_SALARY, MAX(E.SALARY) FROM EMPLOYEES E
    LEFT JOIN DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
    WHERE E.DEPARTMENT_ID IS NOT NULL
    GROUP BY D.DEPARTMENT_NAME, E.JOB_ID
    HAVING ROUND(AVG(E.SALARY),2) > 8000
    ORDER BY AVG_SALARY;
    
    
    SELECT (
        SELECT DEPARTMENT_NAME FROM DEPARTMENTS D 
        WHERE E.DEPARTMENT_ID = D.DEPARTMENT_ID
        ) AS DEPARTMENT_NAME, 
    E.JOB_ID, ROUND(AVG(E.SALARY),2) AS AVG_SALARY, MAX(E.SALARY) FROM EMPLOYEES E
    WHERE E.DEPARTMENT_ID IS NOT NULL
    GROUP BY E.DEPARTMENT_ID, E.JOB_ID
    HAVING ROUND(AVG(E.SALARY),2) > 8000
    ORDER BY AVG_SALARY;
    
    
    --Q2
    SELECT * FROM EMPLOYEES;
    SELECT E1.DEPARTMENT_ID, E1.N, E2.A FROM (
        SELECT DEPARTMENT_ID, CEIL(AVG(SALARY)) AS N FROM EMPLOYEES
        WHERE MONTHS_BETWEEN(SYSDATE, HIRE_DATE)<= 60
        GROUP BY DEPARTMENT_ID
    ) E1
    LEFT JOIN(
        SELECT DEPARTMENT_ID, CEIL(AVG(SALARY)) AS A FROM EMPLOYEES
        GROUP BY DEPARTMENT_ID
    ) E2
    ON E1.DEPARTMENT_ID = E2.DEPARTMENT_ID
    WHERE E1.N>=E2.A;
    
    
    -- Q3
    SELECT C.NAME, L.AMOUNT, L.STATUS,B.LOCATION FROM BRANCHES B
    INNER JOIN LOANS L  ON B.BRANCH_ID = L.BRANCH_ID AND B.LOCATION = 'Location 4' AND L.AMOUNT >=50000
    LEFT JOIN CUSTOMERS C ON L.CUSTOMER_ID = C.CUSTOMER_ID;
    
    
    --Q4
    SELECT C.NAME, C.EMAIL, C.PHONE FROM CUSTOMERS C
    INNER JOIN(
        SELECT CUSTOMER_ID FROM LOANS
        WHERE STATUS = 'PENDING'
        INTERSECT
        SELECT CUSTOMER_ID FROM LOANS
        WHERE STATUS = 'REJECTED'
    ) L ON C.CUSTOMER_ID  = L.CUSTOMER_ID ;
    
    -- Q5
    SELECT C.NAME AS "고객 이름", A.BALANCE AS "금액" FROM CUSTOMERS C
    INNER JOIN ACCOUNTS A ON C.CUSTOMER_ID = A.CUSTOMER_ID AND A.BALANCE>500000
    UNION ALL
    SELECT C.NAME AS "고객 이름", L.AMOUNT AS "금액" FROM CUSTOMERS C
    INNER JOIN LOANS L ON C.CUSTOMER_ID = L.CUSTOMER_ID AND L.STATUS = 'PENDING'
    ORDER BY "고객 이름";
   ```
 - 윈도우 함수
  ```
    -- 관련 실습
    SELECT EMPLOYEE_ID, NAME, SALARY,
        RANK() OVER( ORDER BY SALARY DESC) AS RANK 
        FROM EMPLOYEES;
    
    
    SELECT DEPARTMENT_ID, EMPLOYEE_ID, NAME, SALARY,
        RANK() OVER( PARTITION BY DEPARTMENT_ID ORDER BY SALARY DESC) AS RANK 
        FROM EMPLOYEES;
        
    SELECT EMPLOYEE_ID NAME, SALARY,
        DENSE_RANK()OVER (ORDER BY SALARY DESC)AS RANK FROM EMPLOYEES;
        
    SELECT EMPLOYEE_ID, NAME, SALARY,
        ROW_NUMBER() OVER(ORDER BY SALARY DESC) AS RANK
        FROM EMPLOYEES;
    
    SELECT EMPLOYEE_ID, DEPARTMENT_ID, SALARY,
        SUM(SALARY) OVER
        (PARTITION BY DEPARTMENT_ID) AS DEPT_TOTAL
    FROM EMPLOYEES;
    
    SELECT EMPLOYEE_ID, DEPARTMENT_ID, SALARY,
        SUM(SALARY) OVER
        (PARTITION BY DEPARTMENT_ID ORDER BY SALARY)AS DEPT_TOTAL
    FROM EMPLOYEES;
    
    SELECT EMPLOYEE_ID, DEPARTMENT_ID, SALARY,
        AVG(SALARY) OVER
        (PARTITION BY DEPARTMENT_ID) AS DEPT_AVG
    FROM EMPLOYEES;
    
    SELECT * FROM
        (  
        SELECT EMPLOYEE_ID, DEPARTMENT_ID, SALARY,
            MAX(SALARY) OVER
            (PARTITION BY DEPARTMENT_ID) AS DEPT_MAX
        FROM EMPLOYEES
        )
    WHERE SALARY< DEPT_MAX;
    
    SELECT EMPLOYEE_ID, DEPARTMENT_ID, SALARY,
        MIN(SALARY) OVER
        (PARTITION BY DEPARTMENT_ID) AS DEPT_MIN
    FROM EMPLOYEES;
    
    SELECT EMPLOYEE_ID, DEPARTMENT_ID,
        COUNT(EMPLOYEE_ID) OVER
        (PARTITION BY DEPARTMENT_ID) AS DEPT_CNT
    FROM EMPLOYEES;
  ```
