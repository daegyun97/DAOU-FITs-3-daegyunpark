  @FXML
  private TableView<Book> tableView;
  @FXML
  private TableColumn<Book, String> isbn;
  @FXML
  private TableColumn<Book, String> title;
  @FXML
  private TableColumn<Book, Integer> price;
  @FXML
  private TableColumn<Book, String> author;
  @FXML
  private Button btn;
  @FXML
  private TextField input;

  BookService bookService = new BookService();

  public TableViewController() throws Exception {
  }

  @Override
  public void initialize(URL url, ResourceBundle resourceBundle) {
    isbn.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
    title.setCellValueFactory(new PropertyValueFactory<>("btitle"));
    price.setCellValueFactory(new PropertyValueFactory<>("bprice"));
    author.setCellValueFactory(new PropertyValueFactory<>("bauthor"));
    author.setCellFactory(TextFieldTableCell.forTableColumn());
    author.setOnEditCommit(event -> { // 수정된 값 저장
      Book book = event.getRowValue();
      book.setBauthor(event.getNewValue());
      try {
        bookService.updateBook(event.getRowValue());
      } catch (SQLException e) {
        throw new RuntimeException(e);
      }
    });
    tableView.setOnMouseClicked(event -> {
      tableView.setEditable(true);
    });
/////////////////////////////////////////////////////
    btn.setOnAction(event->{

      String keyWord = input.getText();
      try {
        // ObservableList<Book>로 변환
        ObservableList<Book> books = FXCollections.observableArrayList(bookService.getBooks("%"+keyWord+"%"));
        tableView.setItems(books);
      } catch (SQLException e) {
        throw new RuntimeException(e);
      }
    });

////////////////////////////////////////////////////////
    // ContextMenu (우클릭 메뉴) 생성
    ContextMenu contextMenu = new ContextMenu();
    // 삭제 메뉴 항목
    MenuItem deleteItem = new MenuItem("삭제");
    // ContextMenu에 삭제 메뉴 항목 추가
    contextMenu.getItems().add(deleteItem);
    // 테이블에서 우클릭으로 ContextMenu 띄우기
    tableView.setContextMenu(contextMenu);
    deleteItem.setOnAction(e -> {
      // 우클릭한 항목 가져오기
      Book selectedBook = tableView.getSelectionModel().getSelectedItem();
      if (selectedBook != null) {
        try {
          System.out.println(selectedBook.getBisbn());
          bookService.deleteBook(selectedBook.getBisbn());
        } catch (SQLException ex) {
          throw new RuntimeException(ex);
        }
        tableView.getItems().remove(selectedBook);
      }
    });



  }
}

package booksearch;

import booksearch.VO.Book;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;

public class BookRepository {

  private Connection con = null;
  PreparedStatement ps = null;

  public BookRepository() throws Exception {
    Class.forName("oracle.jdbc.driver.OracleDriver");
    String URL = "jdbc:oracle:thin:@localhost:1521:xe";
    con = DriverManager.getConnection(URL, "C##TEST1", "0000");
    PreparedStatement ps = null;
  }

  public List<Book> findAll(String keyWord) throws SQLException {
    String sql = "select bisbn, btitle, bauthor, bprice from book"
        + " where btitle like ?";
    ps = con.prepareStatement(sql);
    ps.setString(1, keyWord);
    List<Book> books = new ArrayList<>();
    ResultSet rs = ps.executeQuery();
    while (rs.next()) {
      books.add(new Book(rs.getString("bisbn"),rs.getString("btitle"),
          rs.getInt("bprice"),rs.getString("bauthor")));
    }
    return books;
  }

  public void updateById(Book book) throws SQLException {
    String sql = "UPDATE BOOK "
        + "SET bauthor = ?, "
        + "btitle = ?, "
        + "bprice = ? "
        + "WHERE bisbn = ?";
    ps = con.prepareStatement(sql);
    ps.setString(1, book.getBauthor());
    ps.setString(2, book.getBtitle());
    ps.setInt(3, book.getBprice());
    ps.setString(4, book.getBisbn());
    ps.executeUpdate();
  }

  public void deleteByKey(String bisbn) throws SQLException {
    String sql = "delete from book where bisbn=?";
    ps = con.prepareStatement(sql);
    ps.setString(1,bisbn);
    ps.executeUpdate();
  }

}
