SELECT COUNT(DISTINCT E.EMP_NO) AS CNT
FROM 
    ( SELECT EMP_NO
        FROM SALARIES
        WHERE SALARY > 50000
    ) S, EMPLOYEES E
WHERE E.EMP_NO = S.EMP_NO;

SELECT COUNT(E.EMP_NO) AS CNT
FROM EMPLOYEES E
WHERE EXISTS
    ( SELECT 1
        FROM SALARIES S
        WHERE E.EMP_NO = S.EMP_NO AND SALARY > 50000
    );
