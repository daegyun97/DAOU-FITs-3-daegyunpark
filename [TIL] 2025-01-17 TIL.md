### 학습 내용
SQL 쿼리 문제 반복을 통한 구문 작성 실습

<br>
### 실습 상세 내용

```
-- Q1
SELECT  E.NAME,
    CASE WHEN H.JOB_ID IS NULL THEN 'NO CHANGE'
        ELSE H.JOB_ID
        END AS JOB_ID
FROM EMPLOYEES E
LEFT JOIN JOB_HISTORY H ON E.EMPLOYEE_ID = H.EMPLOYEE_ID;

-- Q2
SELECT E.NAME, NVL(E.SALARY, 0) AS SALARY, D.DEPARTMENT_NAME 
FROM EMPLOYEES E
LEFT JOIN  DEPARTMENTS D ON E.DEPARTMENT_ID =D.DEPARTMENT_ID
WHERE E.NAME LIKE '%Employee 1%'
ORDER BY D.DEPARTMENT_NAME;

-- Q3
SELECT T.ACCOUNT_ID, T.AMOUNT, TO_CHAR(T.TRANSACTION_DATE,'YYYY-MM-DD HH:mm:SS'), A.CUSTOMER_ID, C.NAME
FROM TRANSACTIONS T 
LEFT JOIN ACCOUNTS A ON A.ACCOUNT_ID = T.ACCOUNT_ID
LEFT JOIN CUSTOMERS C ON A.CUSTOMER_ID = C.CUSTOMER_ID
WHERE MONTHS_BETWEEN(SYSDATE , T.TRANSACTION_DATE)<=6;

-- Q4
SELECT E.JOB_ID, COUNT(*) FROM DEPARTMENTS D
LEFT JOIN  EMPLOYEES E ON D.DEPARTMENT_ID = E.DEPARTMENT_ID
WHERE D.DEPARTMENT_NAME IN ('Department 3','Department 5','Department 4','Department 6')
GROUP BY E.JOB_ID;

-- Q5
SELECT * FROM ACCOUNTS;
SELECT ACCOUNT_TYPE AS "계좌 유형", TRUNC(AVG(BALANCE),2) AS "평균 잔액" FROM ACCOUNTS
GROUP BY ACCOUNT_TYPE;
ORDER BY 평균 잔액;

-- Q6
SELECT A.ACCOUNT_ID, COUNT(TRANSACTION_ID) AS CNT
FROM ACCOUNTS A
LEFT JOIN TRANSACTIONS T ON A.ACCOUNT_ID = T.ACCOUNT_ID
GROUP BY A.ACCOUNT_ID
ORDER BY CNT;

-- Q7
SELECT * FROM
(
    SELECT C.NAME, SUM(T.AMOUNT) AS TA
    FROM TRANSACTIONS T
    JOIN ACCOUNTS A ON A.ACCOUNT_ID = T.ACCOUNT_ID
    JOIN CUSTOMERS C ON A.CUSTOMER_ID = C.CUSTOMER_ID
    WHERE T.TRANSACTION_DATE BETWEEN '2024-01-01' AND '2025-12-31'
    GROUP BY C.NAME
    ORDER BY TA DESC
)
WHERE ROWNUM <11;

-- Q8
SELECT E.NAME, D.DEPARTMENT_NAME FROM EMPLOYEES E
LEFT JOIN DEPARTMENTS D ON D.DEPARTMENT_ID = E.DEPARTMENT_ID
WHERE E.SALARY IS NULL;


-- Q9
SELECT * FROM
    (
        SELECT CUSTOMER_ID, MAX(AMOUNT) FROM LOANS
        GROUP BY CUSTOMER_ID
        ORDER BY MAX(AMOUNT) DESC
    )
WHERE ROWNUM < 16;

-- Q10
SELECT DEPARTMENT_ID, COUNT(*)
FROM EMPLOYEES 
WHERE (SALARY >=7000 AND SALARY<=15000) AND DEPARTMENT_ID IS NOT NULL
GROUP BY DEPARTMENT_ID
HAVING COUNT(*)>= 5
ORDER BY COUNT(*) DESC;
```
