### 오늘 내가 새롭게 배운 것은 무엇인지?
 - 윈도우 함수
   ```
   SELECT EMPLOYEE_ID, SALARY,
        SUM(SALARY) OVER (ORDER BY SALARY ROWS 2 PRECEDING) AS SALARY_SUM
    FROM EMPLOYEES;
    
    SELECT EMPLOYEE_ID, SALARY,
        SUM(SALARY) OVER (ORDER BY SALARY RANGE UNBOUNDED PRECEDING) AS SALARY_SUM
    FROM EMPLOYEES;
    
    SELECT EMPLOYEE_ID, SALARY, 
        SUM(SALARY) OVER (ORDER BY SALARY ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS FUTURE_SUM
    FROM EMPLOYEES;
    
    SELECT EMPLOYEE_ID, SALARY,
        SUM(SALARY) OVER (ORDER BY SALARY ROWS BETWEEN CURRENT ROW AND 2 FOLLOWING) AS SUM_NEXT_2
    FROM EMPLOYEES;
    
    SELECT EMPLOYEE_ID, DEPARTMENT_ID, SALARY,
        SUM(SALARY) OVER (PARTITION BY DEPARTMENT_ID ORDER BY SALARY ROWS UNBOUNDED PRECEDING) AS CUM_SALARY
    FROM EMPLOYEES;
    
    SELECT CUSTOMER_ID, ACCOUNT_ID, BALANCE, 
        SUM(BALANCE) OVER(PARTITION BY CUSTOMER_ID ORDER BY ACCOUNT_ID) AS CUM_BALANCE
    FROM ACCOUNTS;
   ```

 - 복습
   ```
   -- Q1
    SELECT * FROM
        (
        SELECT JOB_ID, EMPLOYEE_ID, NAME, SALARY,
            ROW_NUMBER() OVER(PARTITION BY JOB_ID ORDER BY SALARY DESC) AS RANKING
        FROM EMPLOYEES
        WHERE SALARY IS NOT NULL
        ORDER BY JOB_ID, RANKING
        )
    WHERE RANKING <= 3;
    
    --Q2
    
    SELECT YEAR, TOTAL,
        SUM(TOTAL) OVER(ORDER BY TOTAL) AS PREFIX
    FROM 
    (
        SELECT EXTRACT(YEAR FROM HIRE_DATE) AS YEAR, SUM(SALARY) AS TOTAL
        FROM EMPLOYEES
        GROUP BY EXTRACT(YEAR FROM HIRE_DATE)
    );
    
    -- Q3
    SELECT EMPLOYEE_ID, NAME, SALARY, 
        (SELECT AVG(SALARY) FROM EMPLOYEES) AS AVG_SAL,
        RANK() OVER(ORDER BY SALARY) AS SALARY_RANK,
        CASE WHEN SALARY>(SELECT AVG(SALARY) FROM EMPLOYEES) THEN 'HIGH'
        ELSE 'LOW'
        END AS R
    FROM EMPLOYEES;
    
    
    -- Q4
    SELECT * FROM
        (
        SELECT DISTINCT CUSTOMER_ID, 
            SUM(AMOUNT) OVER(PARTITION BY CUSTOMER_ID) AS AMOUNT
        FROM LOANS
        WHERE MONTHS_BETWEEN(SYSDATE,LOAN_DATE)<=12
        ORDER BY AMOUNT DESC
        )
    WHERE ROWNUM<=5;
    
    -- Q5
    SELECT NAME, EMPLOYEE_ID,SALARY,
        ROUND(AVG(SALARY) OVER(ORDER BY EMPLOYEE_ID ROWS BETWEEN 2 PRECEDING AND CURRENT ROW),2) AS "이동평균"
    FROM EMPLOYEES;
    
    -- Q6
    SELECT CUSTOMER_ID,AMOUNT,IS_ACCOUNT  FROM
        (
        SELECT CUSTOMER_ID, BALANCE AS AMOUNT,(SELECT AVG(BALANCE) FROM ACCOUNTS)AS AVG_AMOUNT,1 IS_ACCOUNT
        FROM ACCOUNTS
        UNION
        SELECT CUSTOMER_ID, AMOUNT,(SELECT AVG(AMOUNT) FROM ACCOUNTS),0 IS_ACCOUNT
        FROM LOANS
        )
    WHERE AMOUNT > AVG_AMOUNT;
    
    -- Q7
    SELECT * FROM(
        SELECT DISTINCT DEPARTMENT_ID, JOB_ID,
            ROUND(AVG(SALARY) OVER(PARTITION BY DEPARTMENT_ID))AS DEPT, 
            ROUND(AVG(SALARY) OVER(PARTITION BY JOB_ID))AS JOB
        FROM EMPLOYEES
        )
    WHERE JOB>DEPT AND DEPARTMENT_ID IS NOT NULL;
    
    -- Q8
    SELECT BRANCH_ID, COUNT(*) AS CNT, ROUND(COUNT(CASE WHEN STATUS = 'APPROVED' THEN 1 END)/COUNT(*)*100,2) AS APPROVE_PER
    FROM LOANS
    GROUP BY BRANCH_ID; 
    
    -- Q9
    SELECT NAME, SALARY FROM
        (
        SELECT JOB_ID, NAME,SALARY, 
            RANK() OVER(PARTITION BY JOB_ID ORDER BY SALARY DESC) AS RANKING
        FROM EMPLOYEES
        WHERE SALARY IS NOT NULL
        )
    WHERE RANKING=1;
    
    -- Q10
    SELECT D.DEPARTMENT_NAME, COUNT(E.MANAGER_ID) AS IS_MANAGER, COUNT(*) - COUNT(E.MANAGER_ID) AS NOT_MANAGER
    FROM EMPLOYEES E
    LEFT JOIN DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
    WHERE E.DEPARTMENT_ID IS NOT NULL
    GROUP BY  D.DEPARTMENT_NAME;
   ```
