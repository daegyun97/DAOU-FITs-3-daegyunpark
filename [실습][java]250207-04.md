```java
package javaexam2.plane;

public abstract class Plane {

  private String planeName;
  private int fuelSize;

  public Plane() {
  }

  public Plane(String planeName, int fuelSize) {
    this.planeName = planeName;
    this.fuelSize = fuelSize;
  }

  public String getPlaneName() {
    return planeName;
  }

  public void setPlaneName(String planeName) {
    this.planeName = planeName;
  }

  public int getFuelSize() {
    return fuelSize;
  }

  public void setFuelSize(int fuelSize) {
    this.fuelSize = fuelSize;
  }

  @Override
  public String toString() {
    return this.planeName + " " + this.fuelSize;
  }

  public void refuel(int fuel) {
    this.fuelSize += fuel;
  }

  public abstract void flight(int distance);
}

```

```java
package javaexam2.plane;

public class Airplane extends Plane {

  public Airplane() {
  }


  public Airplane(String planeName, int fuelSize) {
    super(planeName, fuelSize);
  }

  @Override
  public void flight(int distance) {
    this.setFuelSize(this.getFuelSize() - (distance * 3));
  }
}
```

```java
package javaexam2.plane;

public class PlaneTest {

  public static void main(String[] args) {
    Plane[] arr = new Plane[]{new Airplane("L747", 1000),
        new Cargoplane("C40", 1000),};

    planeInfo(arr);

    System.out.println("100 운행");
    for (Plane p : arr) {
      p.flight(100);
    }
    planeInfo(arr);

    System.out.println("200 주류");
    for (Plane p : arr) {
      p.refuel(200);
    }
    planeInfo(arr);
  }

  public static void planeInfo(Plane[] arr) {
    System.out.println("Plane fuelSize");
    System.out.println("--------------");
    for (Plane p : arr) {
      System.out.println(p);
    }
    System.out.println();
  }
}
```
