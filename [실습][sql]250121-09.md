-- 실습과제 10
-- Q1
SELECT * FROM TRANSACTIONS;
SELECT ACCOUNT_ID, AMOUNT,TRANSACTION_DATE,
    SUM(AMOUNT) OVER (PARTITION BY ACCOUNT_ID ORDER BY TRANSACTION_DATE ROWS UNBOUNDED PRECEDING) AS PREFIX
FROM TRANSACTIONS;

-- Q2
SELECT ACCOUNT_ID, 
    CONCAT(ROUND(AMOUNT/MAX(AMOUNT) OVER (PARTITION BY ACCOUNT_ID)*100,1),'%') AS "거래 비율"
FROM TRANSACTIONS
ORDER BY ACCOUNT_ID, "거래 비율";

-- Q3
SELECT * FROM LOANS;
SELECT BRANCH_ID , AMOUNT, LOAN_DATE,
    SUM(AMOUNT) OVER(PARTITION BY BRANCH_ID ORDER BY LOAN_DATE ROWS UNBOUNDED PRECEDING) AS PREFIX
FROM LOANS;

-- Q4
SELECT * FROM ACCOUNTS;
SELECT DISTINCT CUSTOMER_ID, 
    ROUND(AVG(BALANCE) OVER(PARTITION BY CUSTOMER_ID),2) AS AVG_AMOUNT
FROM ACCOUNTS
ORDER BY CUSTOMER_ID;
