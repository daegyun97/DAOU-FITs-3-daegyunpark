package first;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;

public class TransactionTest {

  static final String DRIVER = "oracle.jdbc.driver.OracleDriver";
  static final String IP = "127.0.0.1";
  static final String PORT = "1521";
  static final String SID = "xe";
  static final String USER = "C##TEST1";
  static final String PW = "0000";

  public static void main(String[] args) throws Exception {
    //1. JDBC driver loading
    Class.forName(DRIVER);

    //2.connect to database
    String URL = "jdbc:oracle:thin:@" + IP + ":" + PORT + "/" + SID;
    Connection con = DriverManager.getConnection(URL, USER, PW);
    con.setAutoCommit(false);// 트랜잭션 관리

    //3.create sql statement
    String sql = "INSERT INTO EMPLOYEES (EMP_ID, NAME) "
        + "VALUES (?,?, )"; //인파라미터 사용
    PreparedStatement ps = con.prepareStatement(sql);
    ps.setString(1, "1");
    ps.setString(2, "Gildong");

    //4.execute
    int rs = ps.executeUpdate(sql);

    //5.result
    con.commit();//트랜잭션 완료
    //con.rollback(); 요건 롤백처리

    //6.disconnect database
    ps.close();
    con.close();
  }
}
