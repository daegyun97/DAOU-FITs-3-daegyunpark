-- Q1
SELECT DEPARTMENT_ID, SUM(SALARY)
FROM EMPLOYEES
WHERE DEPARTMENT_ID IS NOT NULL
GROUP BY ROLLUP(DEPARTMENT_ID);

-- Q2
SELECT HIRE_DATE, COUNT(*) AS CNT
FROM EMPLOYEES
GROUP BY ROLLUP(HIRE_DATE)
HAVING COUNT(*) >= 2;

-- Q3
SELECT NVL(B.NAME,'ALL BRANCHES') AS BRANCH, SUM(L.AMOUNT)
FROM LOANS L
LEFT JOIN BRANCHES B ON B.BRANCH_ID = L.BRANCH_ID
GROUP BY ROLLUP(B.NAME)
ORDER BY GROUPING(B.NAME), BRANCH ;
